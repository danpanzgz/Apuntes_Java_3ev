public List<Viajero> obtenerTodos() throws BDException {
    List<Viajero> lista = new ArrayList<>();
    String sql = "SELECT codigo, nombre, puntos FROM viajero";

    try (Connection con = ConfigBDSQLite.abrirConexion();
         Statement st = con.createStatement();
         ResultSet rs = st.executeQuery(sql)) {

        while (rs.next()) {
            lista.add(new Viajero(
                rs.getInt("codigo"),
                rs.getString("nombre"),
                rs.getInt("puntos")

        }

    } catch (SQLException e) {
        throw new BDException("Error al obtener viajeros: " + e.getMessage());
    }
    return lista;
}

// Ejemplo: viajeros que han pasado por una estación en una fecha (DISTINCT + ORDER)
public List<Viajero> obtenerPorEstacionYFecha(int codEstacion, String fecha)
        throws BDException {

    List<Viajero> lista = new ArrayList<>();

    String sql = "SELECT DISTINCT v.codigo, v.nombre, v.puntos " +
                 "FROM viajero v " +
                 "JOIN billete b ON v.codigo = b.codigo_viajero " +
                 "WHERE (b.codigo_estacion_origen = ? OR b.codigo_estacion_destino = ?) " +
                 "AND b.fecha = ? " +
                 "ORDER BY v.nombre ASC";

    try (Connection con = ConfigBDSQLite.abrirConexion();
         PreparedStatement ps = con.prepareStatement(sql)) {

        ps.setInt(1, codEstacion);
        ps.setInt(2, codEstacion);
        ps.setString(3, fecha);
        ResultSet rs = ps.executeQuery();

        while (rs.next()) {
            lista.add(new Viajero(
                rs.getInt("codigo"),
                rs.getString("nombre"),
                rs.getInt("puntos")
            ));
        }

    } catch (SQLException e) {
        throw new BDException("Error: " + e.getMessage());
    }
    return lista;
}

public int insertar(int codViajero, String fecha, double importe)
        throws BDException {

    String sql = "INSERT INTO billete (codigo_viajero, fecha, importe) VALUES (?, ?, ?)";

    try (Connection con = ConfigBDSQLite.abrirConexion();
         PreparedStatement ps = con.prepareStatement(sql, Statement.RETURN_GENERATED_KEYS)) {

        ps.setInt(1, codViajero);
        ps.setString(2, fecha);
        ps.setDouble(3, importe);

        int filas = ps.executeUpdate();
        if (filas == 0) throw new BDException("No se insertó ninguna fila.");

        ResultSet keys = ps.getGeneratedKeys();
        if (keys.next()) return keys.getInt(1); // código generado

    } catch (SQLException e) {
        throw new BDException("Error al insertar: " + e.getMessage());
    }
    throw new BDException("No se obtuvo el código generado.");
}

public List<Clase> leerClases() throws BDException {
    List<Clase> lista = new ArrayList<>();

    try (BufferedReader br = new BufferedReader(new FileReader(rutaFichero))) {
        String linea;
        while ((linea = br.readLine()) != null) {
            linea = linea.trim();
            if (linea.isEmpty()) continue;

            String[] partes = linea.split(";");
            if (partes.length < 4) continue; // línea incompleta

            int codigo = Integer.parseInt(partes[0].trim());
            String nombre = partes[1].trim();
            double consumo = Double.parseDouble(partes[2].trim());
            int porcentaje = (int) Double.parseDouble(partes[3].trim());

            lista.add(new Clase(codigo, nombre, consumo, porcentaje));
        }
    } catch (IOException e) {
        throw new BDException("Error al leer fichero: " + e.getMessage());
    }
    return lista;
}
Escribir List → CSV
public void escribirClases(List<Clase> clases) throws BDException {
    try (BufferedWriter bw = new BufferedWriter(new FileWriter(rutaFichero))) {
        for (Clase c : clases) {
            bw.write(c.getCodigo() + ";" + c.getNombre() + ";" +
                    c.getConsumoMinimoAnual() + ";" + c.getPorcentajeAdicional());
            bw.newLine();
        }
    } catch (IOException e) {
        throw new BDException("Error al escribir fichero: " + e.getMessage());
    }
}
9. Sincronizar CSV con BD Fichero BD
Reglas: Actualiza las que coinciden por código. Elimina las del fichero que no están en BD. Añade las de BD que no están en fichero.
actualizarFicheroDesdeDB()
public void actualizarFicheroDesdeDB() throws BDException {
    AccesoClase accesoClase = new AccesoClase();

    List<Clase> clasesDB = accesoClase.obtenerTodas();    // fuente de verdad
    List<Clase> clasesFichero = leerClases();            // estado actual
    List<Clase> resultado = new ArrayList<>();

    // PASO 1: recorrer fichero → actualizar si existe en BD, descartar si no
    for (Clase cf : clasesFichero) {
        Clase deBD = null;
        for (Clase cdb : clasesDB) {
            if (cdb.getCodigo() == cf.getCodigo()) { deBD = cdb; break; }
        }
        if (deBD != null) {
            cf.setConsumoMinimoAnual(deBD.getConsumoMinimoAnual());
            cf.setPorcentajeAdicional(deBD.getPorcentajeAdicional());
            resultado.add(cf);           // ✅ actualizada
        }                                  // si deBD==null → se descarta ❌
    }

    // PASO 2: añadir de BD las que no estaban en el fichero
    for (Clase cdb : clasesDB) {
        boolean yaExiste = false;
        for (Clase r : resultado) {
            if (r.getCodigo() == cdb.getCodigo()) { yaExiste = true; break; }
        }
        if (!yaExiste) resultado.add(cdb);   // ✅ añadida nueva
    }

    escribirClases(resultado);
    System.out.println("Fichero actualizado. Total: " + resultado.size());
}

private static final Scanner SC = new Scanner(System.in);

public static void main(String[] args) {
    int opcion;
    do {
        System.out.println("\n1. Opción 1\n2. Opción 2\n0. Salir");
        opcion = leerEntero("Elige: ");
        switch (opcion) {
            case 1 -> opcion1();
            case 2 -> opcion2();
            case 0 -> System.out.println("Saliendo...");
            default -> System.out.println("Opción no válida.");
        }
    } while (opcion != 0);
}

// Leer entero seguro (no rompe con letras)
private static int leerEntero(String mensaje) {
    while (true) {
        System.out.print(mensaje);
        try {
            return Integer.parseInt(SC.nextLine().trim());
        } catch (NumberFormatException e) {
            System.out.println("Introduce un número válido.");
        }
    }
}

public class VentanaConsulta extends JFrame {

    private JTextField txtCodigo;    // añadir un campo por dato a leer
    private JTextField txtFecha;
    private JButton btnConsultar;
    private JButton btnCancelar;

    public VentanaConsulta() {
        setTitle("Consulta");
        setDefaultCloseOperation(JFrame.DISPOSE_ON_CLOSE);
        setLayout(new BorderLayout(10, 10));

        // Panel de campos
        JPanel panelCampos = new JPanel(new FlowLayout());
        panelCampos.add(new JLabel("Código:"));
        txtCodigo = new JTextField(8);
        panelCampos.add(txtCodigo);
        panelCampos.add(new JLabel("Fecha:"));
        txtFecha = new JTextField(10);
        panelCampos.add(txtFecha);

        // Panel de botones
        JPanel panelBotones = new JPanel();
        btnConsultar = new JButton("Consultar");
        btnCancelar = new JButton("Cancelar");
        panelBotones.add(btnConsultar);
        panelBotones.add(btnCancelar);

        JPanel norte = new JPanel(new BorderLayout());
        norte.add(panelCampos, BorderLayout.CENTER);
        norte.add(panelBotones, BorderLayout.SOUTH);
        add(norte, BorderLayout.NORTH);

        btnCancelar.addActionListener(e -> dispose());
        btnConsultar.addActionListener(e -> consultar());

        setSize(600, 120);
        setLocationRelativeTo(null);
        setVisible(true);
    }

    private void consultar() {
        int codigo;
        try {
            codigo = Integer.parseInt(txtCodigo.getText().trim());
        } catch (NumberFormatException ex) {
            JOptionPane.showMessageDialog(this, "Código no válido.",
                "Error", JOptionPane.ERROR_MESSAGE);
            return;
        }
        String fecha = txtFecha.getText().trim();

        try {
            AccesoViajero dao = new AccesoViajero();
            List<Viajero> lista = dao.obtenerPorEstacionYFecha(codigo, fecha);

            if (lista.isEmpty()) {
                JOptionPane.showMessageDialog(this,
                    "No se encontraron resultados.",
                    "Error", JOptionPane.ERROR_MESSAGE);
            } else {
                mostrarTabla(lista);
            }

        } catch (BDException ex) {
            JOptionPane.showMessageDialog(this, "Error BD: " + ex.getMessage(),
                "Error", JOptionPane.ERROR_MESSAGE);
        }
    }

    private void mostrarTabla(List<Viajero> lista) {
        JDialog dialogo = new JDialog(this, "Resultados", true);
        dialogo.setLayout(new BorderLayout());

        // ← CAMBIA los nombres de columnas
        String[] columnas = {"Código", "Nombre", "Puntos"};
        DefaultTableModel modelo = new DefaultTableModel(columnas, 0);

        for (Viajero v : lista) {
            // ← CAMBIA los getters según tus atributos
            modelo.addRow(new Object[]{v.getCodigo(), v.getNombre(), v.getPuntos()});
        }

        dialogo.add(new JScrollPane(new JTable(modelo)), BorderLayout.CENTER);

        JButton btnCerrar = new JButton("Cerrar");
        btnCerrar.addActionListener(e -> dialogo.dispose());
        JPanel p = new JPanel(); p.add(btnCerrar);
        dialogo.add(p, BorderLayout.SOUTH);

        dialogo.setSize(600, 300);
        dialogo.setLocationRelativeTo(this);
        dialogo.setVisible(true);
    }
}
12. Clases OOP con contador estático OOP
Patrón típico: código autoincremental (OBR1, OBR2...) + array de datos por día/categoría.
Clase con ID autoincremental + array acumulador
public class Obrera {

    private static int contador = 0;   // compartido por TODAS las instancias
    private String codigo;
    private int[] datos;               // array de 7 enteros (uno por día)

    public Obrera() {
        contador++;
        this.codigo = "OBR" + contador;  // OBR1, OBR2, OBR3...
        this.datos = new int[7];         // inicializado a 0 automáticamente
    }

    // Acumula un valor aleatorio en el índice indicado
    public void conseguirComida(int dia) {
        if (dia < 0 || dia > 6) return;
        datos[dia] += (int) (Math.random() * 10); // 0-9
    }

    // Suma todos los elementos del array
    public int totalAcumulada() {
        int total = 0;
        for (int d : datos) total += d;
        return total;
    }

    public String getCodigo() { return codigo; }
    public int[] getDatos() { return datos; }
}

public class Hormiguero {

    private Obrera[] obreras;
    private int numObreras;

    public Hormiguero(int capacidad) {
        obreras = new Obrera[capacidad];
        numObreras = 0;
    }

    // Añadir elemento al array
    public Obrera añadir() {
        if (numObreras >= obreras.length) { System.out.println("Lleno."); return null; }
        Obrera nueva = new Obrera();
        obreras[numObreras++] = nueva;
        return nueva;
    }

    // Obtener el de mayor valor (patrón máximo)
    public Obrera conMasComida() {
        if (numObreras == 0) return null;
        Obrera max = obreras[0];
        for (int i = 1; i < numObreras; i++) {
            if (obreras[i].totalAcumulada() > max.totalAcumulada()) max = obreras[i];
        }
        return max;
    }

    // Suma total de todos los elementos
    public int totalHormiguero() {
        int total = 0;
        for (int i = 0; i < numObreras; i++) total += obreras[i].totalAcumulada();
        return total;
    }
}
